# ==========================
# Example STA Netlist
# ==========================
CLOCK_PERIOD 10 
# Nodes
NODE FF1_Q FF_Q
NODE FF2_Q FF_Q

NODE U1 GATE
NODE U2 GATE
NODE U3 GATE

NODE FF3_D FF_D
NODE FF4_D FF_D

# Connections
EDGE FF1_Q U1
EDGE U1 U2
EDGE U2 FF3_D

EDGE FF2_Q U3
EDGE U3 FF4_D

# Gate Delays
# DELAY <node> <max> <min>

DELAY U1 2.0 0.5
DELAY U2 3.0 1.0
DELAY U3 6.0 2.0

# Flip-Flop Parameters

CLOCK_TO_Q FF1_Q 0.5
CLOCK_TO_Q FF2_Q 0.5

SETUP FF3_D 0.5
SETUP FF4_D 0.5

HOLD FF3_D 0.2
HOLD FF4_D 0.2
